document.addEventListener('DOMContentLoaded',()=>{const y=document.querySelector('#year'); if(y)y.textContent=new Date().getFullYear(); const b=document.querySelector('#hamb'); const m=document.querySelector('#mobileMenu'); if(b&&m)b.onclick=()=>m.classList.toggle('open'); const search=document.querySelector('#searchInput'); const clear=document.querySelector('#clearSearch'); const products=[...document.querySelectorAll('.product')]; const empty=document.querySelector('#emptyState'); function filter(){if(!search)return; const q=search.value.trim().toLowerCase(); let shown=0; products.forEach(p=>{const ok=!q||p.textContent.toLowerCase().includes(q); p.style.display=ok?'block':'none'; if(ok)shown++;}); if(empty) empty.style.display=q&&shown===0?'block':'none';} search?.addEventListener('input',filter); clear?.addEventListener('click',()=>{search.value='';filter();search.focus()}); const sections=[...document.querySelectorAll('.menu-section')]; const tabs=[...document.querySelectorAll('.tabs a')]; const io=new IntersectionObserver(entries=>{entries.forEach(e=>{if(e.isIntersecting){tabs.forEach(t=>t.classList.toggle('active',t.getAttribute('href')==='#'+e.target.id));}})},{rootMargin:'-35% 0px -60% 0px'}); sections.forEach(s=>io.observe(s));});
